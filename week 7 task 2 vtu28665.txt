import java.io.*;
import java.util.*;

class Student {
}

class Rockstar {
}

class Hacker {
}

public class Solution {

    static int[] count(ArrayList<Object> mylist) {
        int student = 0;
        int rockstar = 0;
        int hacker = 0;

        for (Object obj : mylist) {
            if (obj instanceof Student) {
                student++;
            }
            if (obj instanceof Rockstar) {
                rockstar++;
            }
            if (obj instanceof Hacker) {
                hacker++;
            }
        }

        return new int[]{student, rockstar, hacker};
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        ArrayList<Object> mylist = new ArrayList<Object>();

        for (int i = 0; i < n; i++) {
            String type = sc.next();

            if (type.equals("Student")) {
                mylist.add(new Student());
            } else if (type.equals("Rockstar")) {
                mylist.add(new Rockstar());
            } else if (type.equals("Hacker")) {
                mylist.add(new Hacker());
            }
        }

        int[] result = count(mylist);

        System.out.println(result[0] + " " + result[1] + " " + result[2]);
    }
}
